% ===== NOTATION MACROS =====
% Include via \input{macros} in emnlp_2026.tex (before \begin{document}).
% All paper-wide notation is defined here for consistency.

% ----- Architecture -----
\newcommand{\LN}{\mathrm{LN}}           % LayerNorm
\newcommand{\Attn}{\mathrm{Attn}}       % Attention operator
\newcommand{\FFN}{\mathrm{FFN}}         % Feed-forward network

% ----- Hidden state -----
% \hz{l}{t}  →  z^l_t   (layer l, token t)
% \hl{l}     →  z^l     (layer l, all tokens)
\newcommand{\hz}[2]{\mathbf{z}^{#1}_{#2}}
\newcommand{\hl}[1]{\mathbf{z}^{#1}}

% ----- Residual update -----
% \Du{l}{t}     →  Δ^l_t
% \Dl{l}        →  Δ^l
% \Dup{l}{t}    →  Δ^l_{t,∥}   (parallel component)
% \Duperp{l}{t} →  Δ^l_{t,⊥}  (perpendicular component)
\newcommand{\Du}[2]{\Delta^{#1}_{#2}}
\newcommand{\Dl}[1]{\Delta^{#1}}
\newcommand{\Dup}[2]{\Delta^{#1}_{#2,\parallel}}
\newcommand{\Duperp}[2]{\Delta^{#1}_{#2,\perp}}

% ----- Parallel ratio -----
% \pr{l}{t}  →  r^l_t   (per-token parallel ratio)
% \PR{l}     →  R^l     (layer-level parallel ratio, averaged)
\newcommand{\pr}[2]{r^{#1}_{#2}}
\newcommand{\PR}[1]{R^{#1}}

% ----- Attention map -----
% \A          →  A              (full attention weight matrix)
% \Aij{t}{s}  →  A_{ts}        (attention weight, query t, key s)
% \Adiag{t}   →  A_{tt}        (diagonal / self-attention weight)
\newcommand{\A}{A}
\newcommand{\Aij}[2]{A_{#1#2}}
\newcommand{\Adiag}[1]{A_{#1#1}}

% ----- Method names -----
\newcommand{\xsa}{XSA}

% ----- Component scaling coefficients -----
\newcommand{\spara}{s^{(\parallel)}}
\newcommand{\sperp}{s^{(\perp)}}
\newcommand{\sres}{s^{(\mathrm{res})}}

% ----- Shorthand -----
\newcommand{\ie}{\textit{i.e.},\ }
\newcommand{\eg}{\textit{e.g.},\ }
\newcommand{\etc}{\textit{etc}.}
\newcommand{\aka}{\textit{a.k.a.}\ }
